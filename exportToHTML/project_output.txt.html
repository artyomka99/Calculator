#!/bin/bash

# Конвертер проекта в TXT формат для Git Bash на Windows 11
# Автор: Project to TXT Converter
# Версия: 1.0

# Цвета для вывода
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
WHITE='\033[1;37m'
NC='\033[0m' # No Color

# Настройки по умолчанию
PROJECT_PATH="."
OUTPUT_FILE="project_output.txt"
OUTPUT_DIR="txt_output"
MAX_FILE_SIZE_MB=10
INCLUDE_TREE=true
INCLUDE_CONTENT=true

# Расширения файлов для включения
INCLUDE_EXTENSIONS=(
    ".py" ".js" ".html" ".css" ".java" ".cpp" ".c" ".h"
    ".jsx" ".tsx" ".ts" ".json" ".xml" ".yaml" ".yml"
    ".md" ".txt" ".sh" ".bat" ".ps1" ".sql" ".php" ".rb"

# Папки для исключения
EXCLUDE_DIRS=(
    "node_modules" "__pycache__" ".git" "venv" "env"
    ".venv" "dist" "build" ".idea" ".vscode" "target"
    "bin" "obj" ".pytest_cache" ".tox" "vendor" "packages"

# Бинарные расширения для исключения
BINARY_EXTENSIONS=(
    ".exe" ".dll" ".so" ".dylib" ".pdf" ".doc" ".docx"
    ".xls" ".xlsx" ".ppt" ".pptx" ".jpg" ".jpeg" ".png"
    ".gif" ".bmp" ".ico" ".svg" ".mp3" ".mp4" ".avi"

# Счетчики
FILES_PROCESSED=0
FILES_SKIPPED=0
TOTAL_SIZE=0

# Функция для форматирования размера файла
format_size() {
    local size=$1
    if [ $size -lt 1024 ]; then
        echo "${size} B"
    elif [ $size -lt 1048576 ]; then
        echo "$(( size / 1024 )) KB"
    elif [ $size -lt 1073741824 ]; then
        echo "$(( size / 1048576 )) MB"
    else
        echo "$(( size / 1073741824 )) GB"
    fi
}

# Функция проверки, является ли файл бинарным
is_binary_file() {
    local file="$1"
    local ext="${file##*.}"
    ext=".${ext,,}" # Приводим к нижнему регистру

    for binary_ext in "${BINARY_EXTENSIONS[@]}"; do
        if [[ "$ext" == "$binary_ext" ]]; then
            return 0
        fi
    done
    return 1
}

# Функция проверки, нужно ли исключить директорию
is_excluded_dir() {
    local path="$1"

    for exclude_dir in "${EXCLUDE_DIRS[@]}"; do
        if [[ "$path" == *"/$exclude_dir/"* ]] || [[ "$path" == *"/$exclude_dir" ]]; then
            return 0
        fi
    done
    return 1
}

# Функция проверки, нужно ли включить файл
should_include_file() {
    local file="$1"

    # Проверка на бинарный файл
    if is_binary_file "$file"; then
        return 1
    fi

    # Проверка на исключаемые директории
    if is_excluded_dir "$file"; then
        return 1
    fi

    # Проверка расширения
    local ext="${file##*.}"
    ext=".${ext,,}"

    for include_ext in "${INCLUDE_EXTENSIONS[@]}"; do
        if [[ "$ext" == "$include_ext" ]]; then
            return 0
        fi
    done

    # Если нет расширения, но это текстовый файл (например, Makefile, Dockerfile)
    local basename=$(basename "$file")
    if [[ "$basename" == "Makefile" ]] || [[ "$basename" == "Dockerfile" ]] || [[ "$basename" == "Jenkinsfile" ]]; then
        return 0
    fi

    return 1
}

# Функция для создания дерева проекта
create_tree() {
    local dir="$1"
    local prefix="$2"
    local is_last="$3"

    local items=()
    local dirs=()
    local files=()

    # Собираем все элементы
    while IFS= read -r -d '' item; do
        if [ -d "$item" ]; then
            local basename=$(basename "$item")
            if ! is_excluded_dir "$item"; then
                dirs+=("$item")
            fi
        elif [ -f "$item" ]; then
            if should_include_file "$item"; then
                files+=("$item")
            fi
        fi
    done < <(find "$dir" -maxdepth 1 -mindepth 1 -print0 2>/dev/null | sort -z)

    # Объединяем массивы
    items=("${dirs[@]}" "${files[@]}")

    local count=${#items[@]}
    local index=0

    for item in "${items[@]}"; do
        index=$((index + 1))
        local is_last_item=false
        if [ $index -eq $count ]; then
            is_last_item=true
        fi

        local basename=$(basename "$item")

        if [ -d "$item" ]; then
            if $is_last_item; then
                echo "${prefix}└── 📁 ${basename}/"
                create_tree "$item" "${prefix}    " true
            else
                echo "${prefix}├── 📁 ${basename}/"
                create_tree "$item" "${prefix}│   " false
            fi
        else
            local size=$(stat -f%z "$item" 2>/dev/null || stat -c%s "$item" 2>/dev/null || echo "0")
            local formatted_size=$(format_size $size)
            if $is_last_item; then
                echo "${prefix}└── 📄 ${basename} (${formatted_size})"
            else
                echo "${prefix}├── 📄 ${basename} (${formatted_size})"
            fi
        fi
    done
}

# Функция для безопасного чтения файла
read_file_safely() {
    local file="$1"
    local max_size=$((MAX_FILE_SIZE_MB * 1024 * 1024))

    # Получаем размер файла
    local size=$(stat -f%z "$file" 2>/dev/null || stat -c%s "$file" 2>/dev/null || echo "0")

    if [ $size -gt $max_size ]; then
        echo "[Файл пропущен - размер $(format_size $size) превышает лимит]"
        ((FILES_SKIPPED++))
        return 1
    fi

    # Пробуем прочитать файл
    if file -b --mime-encoding "$file" 2>/dev/null | grep -q "binary"; then
        echo "[Бинарный файл - пропущен]"
        ((FILES_SKIPPED++))
        return 1
    fi

    # Читаем содержимое
    cat "$file" 2>/dev/null || echo "[Ошибка чтения файла]"
    return 0
}

# Функция конвертации в один файл
convert_to_single_file() {
    echo -e "${CYAN}Начинаем конвертацию проекта в один TXT файл...${NC}"

    local project_name=$(basename "$(realpath "$PROJECT_PATH")")
    local project_full_path=$(realpath "$PROJECT_PATH")

    # Начинаем запись в файл
    {
        echo "================================================================================"
        echo "ПРОЕКТ: $project_name"
        echo "ПУТЬ: $project_full_path"
        echo "ДАТА ЭКСПОРТА: $(date '+%Y-%m-%d %H:%M:%S')"
        echo "================================================================================"
        echo ""

        # Структура проекта
        if $INCLUDE_TREE; then
            echo "СТРУКТУРА ПРОЕКТА:"
            echo "--------------------------------------------------------------------------------"
            create_tree "$PROJECT_PATH" "" true
            echo ""
            echo "================================================================================"
            echo ""
        fi

        # Содержимое файлов
        if $INCLUDE_CONTENT; then
            echo "СОДЕРЖИМОЕ ФАЙЛОВ:"
            echo "================================================================================"
            echo ""

            # Находим все файлы
            while IFS= read -r -d '' file; do
                if should_include_file "$file"; then
                    local relative_path="${file#$PROJECT_PATH/}"
                    local size=$(stat -f%z "$file" 2>/dev/null || stat -c%s "$file" 2>/dev/null || echo "0")
                    local modified=$(stat -f "%Sm" -t "%Y-%m-%d %H:%M:%S" "$file" 2>/dev/null || stat -c "%y" "$file" 2>/dev/null | cut -d' ' -f1-2)

                    echo ""
                    echo "================================================================================"
                    echo "ФАЙЛ: $relative_path"
                    echo "РАЗМЕР: $(format_size $size) | ИЗМЕНЕН: $modified"
                    echo "--------------------------------------------------------------------------------"

                    if read_file_safely "$file"; then
                        ((FILES_PROCESSED++))
                        TOTAL_SIZE=$((TOTAL_SIZE + size))
                    fi

                    # Показываем прогресс
                    if [ $((FILES_PROCESSED % 10)) -eq 0 ] && [ $FILES_PROCESSED -gt 0 ]; then
                        echo -e "${YELLOW}  Обработано файлов: $FILES_PROCESSED${NC}" >&2
                    fi
                fi
            done < <(find "$PROJECT_PATH" -type f -print0 2>/dev/null)
        fi

        # Статистика
        echo ""
        echo "================================================================================"
        echo "СТАТИСТИКА:"
        echo "--------------------------------------------------------------------------------"
        echo "Обработано файлов: $FILES_PROCESSED"
        echo "Пропущено файлов: $FILES_SKIPPED"
        echo "Общий размер обработанных файлов: $(format_size $TOTAL_SIZE)"
        echo "================================================================================"
    } > "$OUTPUT_FILE"

    echo -e "${GREEN}✅ Готово!${NC}"
    echo -e "${WHITE}   Файл сохранен: $OUTPUT_FILE${NC}"
    echo -e "${WHITE}   Обработано файлов: $FILES_PROCESSED${NC}"
    echo -e "${WHITE}   Общий размер: $(format_size $TOTAL_SIZE)${NC}"
}

# Функция конвертации в отдельные файлы
convert_to_separate_files() {
    echo -e "${CYAN}Начинаем конвертацию файлов в отдельные TXT...${NC}"

    # Создаем выходную директорию
    mkdir -p "$OUTPUT_DIR"

    FILES_PROCESSED=0

    # Находим все файлы
    while IFS= read -r -d '' file; do
        if should_include_file "$file"; then
            local relative_path="${file#$PROJECT_PATH/}"
            local output_path="$OUTPUT_DIR/${relative_path}.txt"
            local output_dir=$(dirname "$output_path")

            # Создаем директории если нужно
            mkdir -p "$output_dir"

            # Сохраняем содержимое
            {
                echo "Исходный файл: $relative_path"
                echo "================================================================================"
                echo ""
                cat "$file" 2>/dev/null || echo "[Ошибка чтения файла]"
            } > "$output_path"

            ((FILES_PROCESSED++))

            # Показываем прогресс
            if [ $((FILES_PROCESSED % 10)) -eq 0 ]; then
                echo -e "${YELLOW}  Обработано файлов: $FILES_PROCESSED${NC}"
            fi
        fi
    done < <(find "$PROJECT_PATH" -type f -print0 2>/dev/null)

    echo -e "${GREEN}✅ Готово!${NC}"
    echo -e "${WHITE}   Конвертировано файлов: $FILES_PROCESSED${NC}"
    echo -e "${WHITE}   Файлы сохранены в: $OUTPUT_DIR${NC}"
}

# Интерактивный режим
interactive_mode() {
    clear
    echo -e "${CYAN}========================================"
    echo -e "  КОНВЕРТЕР ПРОЕКТА В TXT"
    echo -e "========================================${NC}"
    echo ""

    # Путь к проекту
    echo -e "${YELLOW}Введите путь к проекту (Enter для текущей папки):${NC}"
    read -r path
    if [ -z "$path" ]; then
        PROJECT_PATH="."
    else
        PROJECT_PATH="$path"
    fi

    # Проверяем существование пути
    if [ ! -d "$PROJECT_PATH" ]; then
        echo -e "${RED}Ошибка: Путь '$PROJECT_PATH' не существует!${NC}"
        exit 1
    fi

    echo ""
    echo -e "${YELLOW}Выберите режим конвертации:${NC}"
    echo -e "${WHITE}1. Все файлы в один TXT"
    echo -e "2. Каждый файл в отдельный TXT"
    echo -e "3. Оба варианта${NC}"
    echo ""

    read -r -p "Ваш выбор (1/2/3): " choice

    case $choice in
        1)
            echo ""
            read -r -p "Имя выходного файла (Enter для 'project_output.txt'): " outfile
            if [ -n "$outfile" ]; then
                OUTPUT_FILE="$outfile"
            fi
            convert_to_single_file
            ;;
        2)
            echo ""
            read -r -p "Папка для сохранения (Enter для 'txt_output'): " outdir
            if [ -n "$outdir" ]; then
                OUTPUT_DIR="$outdir"
            fi
            convert_to_separate_files
            ;;
        3)
            convert_to_single_file
            convert_to_separate_files
            ;;
        *)
            echo -e "${RED}Неверный выбор!${NC}"
            exit 1
            ;;
    esac
}

package service;

import com.opencsv.bean.CsvToBeanBuilder;
import com.opencsv.bean.StatefulBeanToCsvBuilder;
import entity.person.Person;
import repository.PersonRepository;
import service.csv.dto.PersonCsvView;

import java.io.File;
import java.io.IOException;
import java.io.Reader;
import java.io.Writer;
import java.nio.file.Files;
import java.util.List;

public class CSVService {
    private final String CSV_FILE = "persons.csv";
    private final PersonRepository personRepository = PersonRepository.getInstance();;

    public void saveToCsv() {
        try (Writer writer = Files.newBufferedWriter(new File(CSV_FILE).toPath())) {
            var beans = personRepository.getPersons().stream().map(PersonCsvView::fromPerson).toList();
            new StatefulBeanToCsvBuilder<PersonCsvView>(writer).build().write(beans);
            System.out.println("Сохранено в CSV.");
        } catch (Exception e) {
            System.out.println("Ошибка при сохранении CSV: " + e.getMessage());
        }
    }

    public void loadFromCsv() {
        try (Reader reader = Files.newBufferedReader(new File(CSV_FILE).toPath())) {
            var csvList = new CsvToBeanBuilder<PersonCsvView>(reader)
                    .withType(PersonCsvView.class)
                    .build()
                    .parse();
            csvList.forEach(view -> personRepository.getPersons().add(view.toPerson()));
            System.out.println("Загружено из CSV.");
        } catch (Exception e) {
            System.out.println("Ошибка при чтении CSV: " + e.getMessage());
        }
    }

    public void printCsvFile() {
        try {
            System.out.println(Files.readString(new File(CSV_FILE).toPath()));
        } catch (IOException e) {
            System.out.println("Ошибка чтения CSV.");
        }
    }
}

package service;

import entity.person.Person;
import entity.pet.*;
import repository.PersonRepository;

import java.util.ArrayList;
import java.util.List;
import java.util.Random;
import java.util.Scanner;

public class PetService {
    private final List<Pet> pets = new ArrayList<>();
    private final PersonRepository personRepository = PersonRepository.getInstance();
    private final Scanner scanner = new Scanner(System.in);



    private final String[] names = {"Барсик", "Мурка", "Шарик", "Том", "Лаки", "Рекс", "Снежок", "Рыжик"};
    private final String[] breeds = {"Сиамский", "Бульдог", "Мейн-кун", "Доберман", "Сфинкс", "Дворовый"};
    private final Random random = new Random();

    public void generatePets() {
        System.out.print("Сколько питомцев хотите создать? ");
        int count = scanner.nextInt();
        scanner.nextLine();
        for (int i = 0; i < count; i++) {
            String petName = names[random.nextInt(names.length)];
            int petAge = random.nextInt(15) + 1;
            String petBreed = breeds[random.nextInt(breeds.length)];
            int type = random.nextInt(4) + 1;

            Pet pet = switch (type) {
                case 1 -> new Dog(petName, petAge, petBreed);
                case 2 -> new Cat(petName, petAge, petBreed);
                case 3 -> new Mouse(petName, petAge, petBreed);
                case 4 -> new Rat(petName, petAge, petBreed);
                default -> null;
            };

            if (pet != null) {
                pets.add(pet);
            }
        }
        System.out.println(count + " питомцев создано.");
    }

    public void printAllPets() {
        if (pets.isEmpty()) {
            System.out.println("Список питомцев пуст.");
            return;
        }

        System.out.println("\nВсе питомцы:");
        for (int i = 0; i < pets.size(); i++) {
            System.out.printf("%d. %s%n", i + 1, pets.get(i));
        }
    }

    public void assignPetsToPersons() {

        if (pets.isEmpty()) {
            System.out.println("Сначала создайте питомцев.");
            return;
        }

        if (personRepository.getPersons().isEmpty()) {
            System.out.println("Сначала создайте людей.");
            return;
        }

        for (Pet pet : pets) {
            Person person = personRepository.getPersons().get(random.nextInt(personRepository.getPersons().size()));
            person.addPet(pet);
            System.out.printf("Питомец %s назначен %s%n", pet.getName(), person.getName());
        }

        pets.clear();
        System.out.println("Все питомцы назначены.");
    }

    public void findPetsByParams() {
        System.out.print("Введите имя (или пусто): ");
        String name = scanner.nextLine().trim();

        System.out.print("Введите возраст (или пусто): ");
        String ageInput = scanner.nextLine().trim();
        Integer age = ageInput.isEmpty() ? null : Integer.parseInt(ageInput);

        System.out.print("Введите породу (или пусто): ");
        String breed = scanner.nextLine().trim();

        List<Pet> found = new ArrayList<>();

        for (Person person : personRepository.getPersons()) {
            for (Pet pet : person.getPets()) {
                boolean matches =
                        (name.isEmpty() || pet.getName().equalsIgnoreCase(name)) &&
                                (age == null || pet.getAge() == age) &&
                                (breed.isEmpty() || pet.getBreed().equalsIgnoreCase(breed));

                if (matches) {
                    found.add(pet);
                }
            }
        }

        if (found.isEmpty()) {
            System.out.println("Ничего не найдено.");
        } else {
            System.out.println("Результаты поиска:");
            found.forEach(System.out::println);
        }
    }
}
